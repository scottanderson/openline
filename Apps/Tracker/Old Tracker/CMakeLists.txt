# HLTracker "Old Tracker" -- the original (pre-1999) Hotline tracker protocol server.
#
# Two Tracker implementations ship in this source tree side by side ("Old Tracker" and
# "New Tracker", see Apps/Tracker/New\ Tracker/CMakeLists.txt) -- they aren't a
# before/after of the same app, they're two different tracker protocol generations
# (the new one adds a login database, multi-tracker list, and stats/servers windows;
# see the diff between their TrackerServ.h files). Nothing in this repo indicates one
# supersedes the other for the purposes of this port, so both are built, as separate
# executables (hltracker_old.exe / hltracker_new.exe) since they have identically-named
# source files that would otherwise collide in a single target.

set(TRACKER_OLD_SOURCES
	Sources/TrackerServ.cp
	Sources/TrackerServViews.cp
	Sources/TrackerServWindows.cp
)
set_source_files_properties(${TRACKER_OLD_SOURCES} PROPERTIES LANGUAGE CXX)

set(TRACKER_OLD_RC Resources/Win/hotlinetrack.rc)

add_executable(hltracker_old WIN32
	${TRACKER_OLD_SOURCES}
	${TRACKER_OLD_RC}
)

target_include_directories(hltracker_old PRIVATE
	Sources
	"Resources/Win"
)
set_source_files_properties(${TRACKER_OLD_RC} PROPERTIES
	INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win;${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win/Icons;${CMAKE_SOURCE_DIR}/AppWarrior/Resources"
)

target_compile_definitions(hltracker_old PRIVATE
	WIN32=1
	_WIN32_WINNT=0x0500
	$<$<BOOL:${HL_NO_QUICKTIME}>:HL_NO_QUICKTIME>
)

target_compile_options(hltracker_old PRIVATE
	-include "${CMAKE_SOURCE_DIR}/AppWarrior/Headers/_AWPrefix(W).h"
)

target_link_libraries(hltracker_old PRIVATE appwarrior)

target_link_libraries(hltracker_old PRIVATE
	ole32 oleaut32 uuid winmm ws2_32 comctl32 gdi32 comdlg32 shell32 wininet advapi32 user32 gdi32
	netapi32
)

target_link_options(hltracker_old PRIVATE -static -static-libgcc -static-libstdc++)

# Old Tracker doesn't call URez::AddProgramFileToSearchChain() anywhere in its Windows
# source, so (unlike Client/Server) it has no runtime .dat file of its own -- only the
# shared AppWarrior/Error Msgs/*.dat files are staged next to the exe.
file(GLOB ERROR_MSGS_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/AppWarrior/Error Msgs/*.dat")

set(_copy_data_script "${CMAKE_CURRENT_BINARY_DIR}/CopyRuntimeData.cmake")
file(WRITE "${_copy_data_script}" "# Auto-generated by Apps/Tracker/Old Tracker/CMakeLists.txt -- copies runtime .dat files next to hltracker_old.exe.\n")
foreach(_f ${ERROR_MSGS_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Error Msgs\")\n")
endforeach()

add_custom_command(TARGET hltracker_old POST_BUILD
	COMMAND ${CMAKE_COMMAND} -DDEST_DIR=$<TARGET_FILE_DIR:hltracker_old> -P "${_copy_data_script}"
	COMMENT "Copying runtime .dat files next to hltracker_old.exe"
)
