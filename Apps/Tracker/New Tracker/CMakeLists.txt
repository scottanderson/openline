# HLTracker "New Tracker" -- the newer Hotline tracker protocol server, adding a login
# database, multi-tracker list, and stats/servers windows over "Old Tracker" (see
# Apps/Tracker/Old\ Tracker/CMakeLists.txt for why both are built rather than picking
# one).

set(TRACKER_NEW_SOURCES
	Sources/TrackerServ.cp
	Sources/TrackerServViews.cp
	Sources/TrackerServWindows.cp
)
set_source_files_properties(${TRACKER_NEW_SOURCES} PROPERTIES LANGUAGE CXX)

set(TRACKER_NEW_RC Resources/Win/hotlinetrack.rc)

add_executable(hltracker_new WIN32
	${TRACKER_NEW_SOURCES}
	${TRACKER_NEW_RC}
)

target_include_directories(hltracker_new PRIVATE
	Sources
	"Resources/Win"
)
set_source_files_properties(${TRACKER_NEW_RC} PROPERTIES
	INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win;${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win/Icons;${CMAKE_SOURCE_DIR}/AppWarrior/Resources"
)

target_compile_definitions(hltracker_new PRIVATE
	WIN32=1
	_WIN32_WINNT=0x0500
	$<$<BOOL:${HL_NO_QUICKTIME}>:HL_NO_QUICKTIME>
)

target_compile_options(hltracker_new PRIVATE
	-include "${CMAKE_SOURCE_DIR}/AppWarrior/Headers/_AWPrefix(W).h"
)

target_link_libraries(hltracker_new PRIVATE appwarrior)

target_link_libraries(hltracker_new PRIVATE
	ole32 oleaut32 uuid winmm ws2_32 comctl32 gdi32 comdlg32 shell32 wininet advapi32 user32 gdi32
	netapi32
)

target_link_options(hltracker_new PRIVATE -static -static-libgcc -static-libstdc++)

# New Tracker calls URez::AddProgramFileToSearchChain("\phltracker.dat") which resolves
# relative to the exe's own folder -- unlike Client/Server this isn't under a "Data/"
# subfolder, it's staged directly next to the exe (matching hltracker.dat's location in
# the source tree). AppWarrior/Error Msgs/*.dat is staged too, as with the other apps.
file(GLOB ERROR_MSGS_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/AppWarrior/Error Msgs/*.dat")

set(_copy_data_script "${CMAKE_CURRENT_BINARY_DIR}/CopyRuntimeData.cmake")
file(WRITE "${_copy_data_script}" "# Auto-generated by Apps/Tracker/New Tracker/CMakeLists.txt -- copies runtime .dat files next to hltracker_new.exe.\n")
file(APPEND "${_copy_data_script}" "file(COPY \"${CMAKE_CURRENT_SOURCE_DIR}/hltracker.dat\" DESTINATION \"\${DEST_DIR}\")\n")
foreach(_f ${ERROR_MSGS_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Error Msgs\")\n")
endforeach()

add_custom_command(TARGET hltracker_new POST_BUILD
	COMMAND ${CMAKE_COMMAND} -DDEST_DIR=$<TARGET_FILE_DIR:hltracker_new> -P "${_copy_data_script}"
	COMMENT "Copying runtime .dat files next to hltracker_new.exe"
)
