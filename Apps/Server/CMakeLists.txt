# HLServer (Hotline Connect server).

set(SERVER_SOURCES
	Source/HotlineServ.cp
	Source/HotlineServIDTranslate.cp
	Source/HotlineServMisc.cp
	Source/HotlineServNewsDatabase.cp
	Source/HotlineServTrans.cp
	Source/HotlineServViews.cp
	Source/HotlineServWindows.cp
	# Shared between client and server; the server needs the folder-download
	# enumeration (CMyDLFldr/CMyDLItem, used by HotlineServTrans.cp) but not the
	# archive decoder (that's client-side only, for unpacking downloaded archives).
	${CMAKE_SOURCE_DIR}/Apps/Common\ Files/HotlineFolderDownload.cp
)
set_source_files_properties(${SERVER_SOURCES} PROPERTIES LANGUAGE CXX)

set(SERVER_RC Resources/Win/hotlineserv.rc)

add_executable(hlserver WIN32
	${SERVER_SOURCES}
	${SERVER_RC}
)

target_include_directories(hlserver PRIVATE
	Source
	"Resources/Win"
	"${CMAKE_SOURCE_DIR}/Apps/Common Files"
)
set_source_files_properties(${SERVER_RC} PROPERTIES
	INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win;${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win/Icons;${CMAKE_SOURCE_DIR}/AppWarrior/Resources"
)

target_compile_definitions(hlserver PRIVATE
	WIN32=1
	_WIN32_WINNT=0x0500
	$<$<BOOL:${HL_NO_QUICKTIME}>:HL_NO_QUICKTIME>
)

# Same prefix-file stand-in as appwarrior/hlclient -- these .cp files also rely on it.
target_compile_options(hlserver PRIVATE
	-include "${CMAKE_SOURCE_DIR}/AppWarrior/Headers/_AWPrefix(W).h"
)

target_link_libraries(hlserver PRIVATE appwarrior)

# System libraries the original Windows build linked against.
target_link_libraries(hlserver PRIVATE
	ole32 oleaut32 uuid winmm ws2_32 comctl32 gdi32 comdlg32 shell32 wininet advapi32 user32 gdi32
	netapi32
)

target_link_options(hlserver PRIVATE -static -static-libgcc -static-libstdc++)

# --- runtime data files -------------------------------------------------------------
# Same story as hlclient: URez::AddProgramFileToSearchChain() resolves *.dat files
# relative to the exe's own folder, so copy Apps/Server/Data/*.dat and the shared
# AppWarrior/Error Msgs/*.dat files next to the built exe.
file(GLOB SERVER_DATA_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Data/*.dat")
file(GLOB ERROR_MSGS_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/AppWarrior/Error Msgs/*.dat")

# See Apps/Client/CMakeLists.txt for why this goes through a generated script rather
# than a direct copy_if_different command (parentheses in Error Msgs filenames).
set(_copy_data_script "${CMAKE_CURRENT_BINARY_DIR}/CopyRuntimeData.cmake")
file(WRITE "${_copy_data_script}" "# Auto-generated by Apps/Server/CMakeLists.txt -- copies runtime .dat files next to hlserver.exe.\n")
foreach(_f ${SERVER_DATA_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Data\")\n")
endforeach()
foreach(_f ${ERROR_MSGS_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Error Msgs\")\n")
endforeach()

add_custom_command(TARGET hlserver POST_BUILD
	COMMAND ${CMAKE_COMMAND} -DDEST_DIR=$<TARGET_FILE_DIR:hlserver> -P "${_copy_data_script}"
	COMMENT "Copying runtime .dat files next to hlserver.exe"
)
