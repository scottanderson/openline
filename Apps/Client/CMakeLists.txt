# HLClient (Hotline Connect chat client).

set(CLIENT_SOURCES
	Source/HotlineTracker.cp
	Source/HotlineMisc.cp
	Source/CChatLog.cp
	Source/HotlineNews.cp
	Source/HotlineNewsReadHistory.cp
	Source/HotlineWindows.cp
	Source/HotlineTasks.cp
	Source/Hotline.cp
	Source/HotlineViews.cp
	Source/Admin/HotlineAdminWindows.cp
	Source/Admin/HotlineAdminViews.cp
	# Shared between client and server; only the client needs them here.
	${CMAKE_SOURCE_DIR}/Apps/Common\ Files/HotlineArchiveDecoder.cp
	${CMAKE_SOURCE_DIR}/Apps/Common\ Files/HotlineFolderDownload.cp
)
set_source_files_properties(${CLIENT_SOURCES} PROPERTIES LANGUAGE CXX)

# .rc resources (icons/cursors/menus/dialogs). windres resolves the bare filenames the
# .rc references (e.g. "19Hotline.ico") relative to -I include dirs, so point it at the
# "Icons" and "Cursors" subdirectories ("Win Icons" also exists but is empty).
set(CLIENT_RC Resources/Win/hotline.rc)

add_executable(hlclient WIN32
	${CLIENT_SOURCES}
	${CLIENT_RC}
)

target_include_directories(hlclient PRIVATE
	Source
	Source/Admin
	"Resources/Win"
	"${CMAKE_SOURCE_DIR}/Apps/Common Files"
)
set_source_files_properties(${CLIENT_RC} PROPERTIES
	INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win;${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win/Icons;${CMAKE_CURRENT_SOURCE_DIR}/Resources/Win/Cursors;${CMAKE_SOURCE_DIR}/AppWarrior/Resources"
)

target_compile_definitions(hlclient PRIVATE
	WIN32=1
	_WIN32_WINNT=0x0500
	$<$<BOOL:${HL_NO_QUICKTIME}>:HL_NO_QUICKTIME>
)

# Same prefix-file stand-in as appwarrior -- these .cp files also rely on it.
target_compile_options(hlclient PRIVATE
	-include "${CMAKE_SOURCE_DIR}/AppWarrior/Headers/_AWPrefix(W).h"
)

target_link_libraries(hlclient PRIVATE appwarrior)

# System libraries the original Windows build linked against.
target_link_libraries(hlclient PRIVATE
	ole32 oleaut32 uuid winmm ws2_32 comctl32 gdi32 comdlg32 shell32 wininet advapi32 user32 gdi32
	netapi32
)

target_link_options(hlclient PRIVATE -static -static-libgcc -static-libstdc++)

# --- runtime data files -------------------------------------------------------------
# Hotline.cp calls URez::AddProgramFileToSearchChain("Data/hlc19.dat") (and friends),
# which resolves relative to the exe's own folder (kProgramFolder) -- so copy
# Apps/Client/Data/*.dat next to the built exe as "Data/". AppWarrior/Error Msgs/*.dat
# files aren't referenced by any path string in the Windows source we could find (they
# may only ever have been loaded via the Mac resource-fork mechanism), but are copied
# next to the exe as well, defensively, under their source-tree name. Only *.dat is
# copied -- the extension-less files alongside them in Error Msgs/ are Mac resource-fork
# remnants from the original checkout, not used by the Windows build.
file(GLOB CLIENT_DATA_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Data/*.dat")
file(GLOB ERROR_MSGS_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/AppWarrior/Error Msgs/*.dat")

# Several of the Error Msgs filenames contain parentheses, e.g. "UError(1).dat". The
# Unix Makefiles generator's COMMAND-argument escaping doesn't handle "(" / ")" when
# invoking copy_if_different directly (breaks with "sh: Syntax error: ( unexpected"),
# so drive the copy from a tiny generated CMake script instead -- script mode
# (cmake -P) takes the file list as real CMake list values, no shell tokenization
# involved.
set(_copy_data_script "${CMAKE_CURRENT_BINARY_DIR}/CopyRuntimeData.cmake")
file(WRITE "${_copy_data_script}" "# Auto-generated by Apps/Client/CMakeLists.txt -- copies runtime .dat files next to hlclient.exe.\n")
foreach(_f ${CLIENT_DATA_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Data\")\n")
endforeach()
foreach(_f ${ERROR_MSGS_FILES})
	file(APPEND "${_copy_data_script}" "file(COPY \"${_f}\" DESTINATION \"\${DEST_DIR}/Error Msgs\")\n")
endforeach()

add_custom_command(TARGET hlclient POST_BUILD
	COMMAND ${CMAKE_COMMAND} -DDEST_DIR=$<TARGET_FILE_DIR:hlclient> -P "${_copy_data_script}"
	COMMENT "Copying runtime .dat files next to hlclient.exe"
)
